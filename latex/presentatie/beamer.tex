        \framesubtitle{Optimization as Feasibility}
        \vspace*{-1.5em} 
        \begin{columns}[onlytextwidth,T]
            % Gradient Based
            \begin{column}{0.48\textwidth}
                \textbf{Gradient Based}
                \begin{equation*}
                    \minimize_{\theta} \quad \mathcal{L}(f(x;\theta), y)
                \end{equation*}
                
                Update step (Gradient Descent):
                \begin{equation*}
                     \theta^{k+1} = \theta^k - \eta \nabla_\theta \mathcal{L}
                \end{equation*}
                
                \vspace{0.5em}
                \centering
                \begin{tikzpicture}[scale=0.8]
                    % Concave function graph
                    \draw[->, gray!50] (-1.8,0) -- (1.8,0); 
                    \draw[->, gray!50] (0,-0.2) -- (0,2.5);
                    \draw[thick, KUL-BLUE, domain=-1.5:1.5, samples=100] plot (\x, {\x*\x});
                    
                    % Point and Gradient Vector
                    \coordinate (x0) at (-1.1, 1.21); % Point at x = -1.1
                    \fill[red] (x0) circle (2.5pt) node[left, font=\scriptsize, text=black, xshift=-2pt] {$\theta^k$};
                    
                    % Tangent/Gradient arrow
                    % Slope at x=-1.1 is 2*(-1.1) = -2.2.
                    % We want to go "downhill", so direction is positive x.
                    % Vector (1, -2.2) normalized roughly. 
                    % Let's just draw a vector pointing down along the curve.
                    \draw[->, >=latex, red, thick] (x0) -- ++(0.5, -1) node[midway, above right, font=\tiny, inner sep=1pt] {$-\nabla \mathcal{L}$};
                    
                    % Dashed drop line to axis (optional, but adds context)
                    \draw[dashed, gray] (x0) -- (-1.1, 0);
                    \node[below, font=\tiny, gray] at (-1.1, 0) {$\theta^k$};

                    \fill[gray] (0,0) circle (1.5pt);
                \end{tikzpicture}
            \end{column}

    % Standard Algorithms Slide
    \begin{frame}[KUL-standard]
        \frametitle{Solving Feasibility Problems \cite{bauschkyProjections}}
        Find $x \in \bigcap C_i$.\\
        Projection on to set: $P_C(x) = \arg\min_{y \in C} \|x - y\|$\\
        {\scriptsize Strong theoretical results in the convex case (eg \cite{doi:10.1137/S0036144593251710}); but not in the non-convex case.\\}

        \vspace{1em}
        \textbf{Common Algorithms:}
        \begin{columns}[onlytextwidth,T]
            \begin{column}{0.3\textwidth}
                \textbf{\color{KUL-BLUE} Alternating Projections (AP)}
                \begin{equation*}
                    x_{k+1} = P_{C_2}(P_{C_1}(x_k))
                \end{equation*}
            \end{column}
            \begin{column}{0.3\textwidth}
                \textbf{\color{KUL-BLUE} Douglas-Rachford (DR)}
                \begin{align*}
                     R_C &= 2P_C - I \quad \\
                    x_{k+1} &= \frac{1}{2}(I + R_{C_2}R_{C_1})x_k
                \end{align*}
            \end{column}
            \begin{column}{0.3\textwidth}
                \textbf{\color{KUL-BLUE} Cyclic, Dykstra , etc.}
                \begin{align*}
                    x_{k+1} =f(x_k)
                \end{align*}
            \end{column}
        \end{columns}
    \end{frame}

    % Bilinear Constraint Section
    % Slide 1: Problem Definition
    \begin{frame}[KUL-standard]
        \frametitle{Projection for the Bilinear constraint \cite{elser2019learning}}
        \framesubtitle{1. Problem Statement}
        \textbf{Constraint:}
        \begin{equation*}
            C = \{ (y, x, w) \mid y = w^\top x \}
        \end{equation*}
        
        \textbf{Projection Objective:}
        Find closest point $(y, x, w) \in \mathbb{R} \times \mathbb{R}^{n} \times \mathbb{R}^{n}$ to the prior estimate $(y_0, x_0, w_0)$ (If $\gamma$ $\to$ $\infty$: $y \to y_0$):\\
        \begin{equation*}
             \minimize_{y, x, w} \quad \|x - x_0\|^2 + \|w - w_0\|^2 + \gamma^2(y - y_0)^2 
        \end{equation*}
        \vspace{0.5em}
    \end{frame}

    % Slide 2: Lagrangian Analysis
    \begin{frame}[KUL-standard]
        \frametitle{Projection for the Bilinear constraint \cite{elser2019learning}}
        \framesubtitle{2. Lagrangian Formulation}
        The Lagrangian with multiplier $\lambda$ is:
        \begin{equation*}
            L = \|x - x_0\|^2 + \|w - w_0\|^2 + \gamma^2(y - y_0)^2 + \lambda( y - w^\top x)
        \end{equation*}
        
        Setting $\nabla L = 0$ yields (with $u = \lambda/2$):
        \begin{columns}[onlytextwidth,T]
            \begin{column}{0.48\textwidth}
                \begin{align*}
                    x &= x_0 + u w \\
                    w &= w_0 + u x
                \end{align*}
            \end{column}
            \begin{column}{0.48\textwidth}
                \begin{align*}
                    y &= y_0 - \frac{ u}{\gamma^2}\\
                    y &= w^\top x
                \end{align*}
            \end{column}
        \end{columns}
        \vspace{1em}
        Solving the coupled system for $x$ and $w$:
        \begin{equation*}
             x = \frac{x_0 + u w_0}{1 - u^2}, \quad w = \frac{w_0 + u x_0}{1 - u^2}
        \end{equation*}
    \end{frame}

    % Slide 3: Solution
    \begin{frame}[KUL-standard]
        \frametitle{Projection for the Bilinear constraint}
        \framesubtitle{3. Analytic Solution}
        Substituting $x(u), w(u), y(u)$ back into the constraint $ y = w^\top x$:
        \begin{equation*}
         \left(y_0 - \frac{u}{\gamma^2}\right) = \frac{(w_0 + u x_0)^\top (x_0 + u w_0)}{(1-u^2)^2}
        \end{equation*}
        
        This simplifies to a scalar root-finding problem $h(u) = 0$:
        \begin{equation*}
            h(u) = \frac{p(1+u^2) + q u}{(1-u^2)^2} -y_0 + \frac{u}{\gamma^2} = 0
        \end{equation*}
        where $p = w_0^\top x_0$ and $q = \|x_0\|^2 + \|w_0\|^2$.
        \vspace{1em}
        
        \textbf{Solution:} Solve for $u$ (e.g., via Newton's method), then reconstruct $x, w, y$.
    \end{frame}
