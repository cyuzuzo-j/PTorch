    \begin{frame}[KUL-standard]
        \frametitle{Paradigm Shift}
        \framesubtitle{Optimization as Feasibility}
        \begin{columns}[onlytextwidth,T]
            % Gradient Based
            \begin{column}{0.48\textwidth}
                \textbf{\large Gradient Based}\\
                \vspace{0.5em}
                Minimize a loss function: % moet technisch gezien cost zijn denk ik 
                \begin{equation*}
                    \min_{\theta} \mathcal{L}(f(x;\theta), y)
                \end{equation*}
                
                Update step (Gradient Descent):
                \begin{equation*}
                    \theta \leftarrow \theta - \eta \nabla_\theta \mathcal{L}
                \end{equation*}
                \vfill
            \end{column}
            
            \hspace{0.5em}

            % Projection Based
            \begin{column}{0.48\textwidth}
                \vspace{-0.1em}
                \textbf{\large Feasibility Based}\\
                \vspace{0.5em}
                Find a point in the intersection of constraints:
                \begin{equation*}
                    x \in \bigcap_{i=1}^m C_i
                \end{equation*}
                
                Update step (e.g. Cyclic Projections):
                \begin{equation*}
                    x \leftarrow P_{C_1}P_{C_2} \ldots P_{C_m}(x)
                \end{equation*}
                \vfill
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}[KUL-standard]
        \frametitle{Solving Feasibility Problems}
        Find $x \in \bigcap C_i$.\\
        Projection on to set: $P_C(x) = \arg\min_{y \in C} \|x - y\|$\\
        
        \vspace{1em}
        \textbf{Common Algorithms:}
        \begin{columns}[onlytextwidth,T]
            \begin{column}{0.3\textwidth}
                \textbf{\color{KUL-BLUE} Alternating Projections (AP)}
                \begin{equation*}
                    x_{k+1} = P_{C_2}(P_{C_1}(x_k))
                \end{equation*}
            \end{column}
            \begin{column}{0.3\textwidth}
                \textbf{\color{KUL-BLUE} Douglas-Rachford (DR)}
                \begin{align*}
                     R_C &= 2P_C - I \quad \\
                    x_{k+1} &= \frac{1}{2}(I + R_{C_2}R_{C_1})x_k
                \end{align*}
            \end{column}
            \begin{column}{0.3\textwidth}
                \textbf{\color{KUL-BLUE} Cyclic, Dykstra , ....}
                \begin{align*}
                    x_{k+1} =(x_k)
                \end{align*}
            \end{column}
        \end{columns}
    \end{frame}

    % Bilinear Constraint Section
    % Bilinear Constraint Section
    % Slide 1: Problem Definition
    \begin{frame}[KUL-standard]
        \frametitle{Projection for the Bilinear constraint}
        \framesubtitle{1. Problem Statement}
        \textbf{Constraint:}
        \begin{equation*}
            C = \{ (y, x, w) \mid y = w^\top x \}
        \end{equation*}
        
        \textbf{Projection Objective:}
        Find the closest point $(y, x, w)$ to the prior estimate $(y_0, x_0, w_0)$:
        \begin{equation*}
             \min_{y, x, w} \quad \|x - x_0\|^2 + \|w - w_0\|^2 + \gamma^2(y - y_0)^2 
        \end{equation*}
        \vspace{0.5em}
    \end{frame}

    % Slide 2: Lagrangian Analysis
    \begin{frame}[KUL-standard]
        \frametitle{Projection for the Bilinear constraint}
        \framesubtitle{2. Lagrangian Formulation}
        The Lagrangian with multiplier $\lambda$ is:
        \begin{equation*}
            L = \|x - x_0\|^2 + \|w - w_0\|^2 + \gamma^2(y - y_0)^2 + \lambda( y - w^\top x)
        \end{equation*}
        
        Setting $\nabla L = 0$ yields the stationarity conditions (with $u = \lambda/2$):
        \begin{columns}[onlytextwidth,T]
            \begin{column}{0.48\textwidth}
                \begin{align*}
                    x &= x_0 + u w \\
                    w &= w_0 + u x
                \end{align*}
            \end{column}
            \begin{column}{0.48\textwidth}
                \begin{align*}
                    y &= y_0 - \frac{ u}{\gamma^2}
                \end{align*}
            \end{column}
        \end{columns}
        \vspace{1em}
        Solving the coupled system for $x$ and $w$:
        \begin{equation*}
             x = \frac{x_0 + u w_0}{1 - u^2}, \quad w = \frac{w_0 + u x_0}{1 - u^2}
        \end{equation*}
    \end{frame}

    % Slide 3: Solution
    \begin{frame}[KUL-standard]
        \frametitle{Projection for the Bilinear constraint}
        \framesubtitle{3. Analytic Solution}
        Substituting $x(u), w(u), y(u)$ back into the constraint $ y = w^\top x$:
        \begin{equation*}
         \left(y_0 - \frac{u}{\gamma^2}\right) = \frac{(w_0 + u x_0)^\top (x_0 + u w_0)}{(1-u^2)^2}
        \end{equation*}
        
        This simplifies to a scalar root-finding problem $h(u) = 0$:
        \begin{equation*}
            h(u) = \frac{p(1+u^2) + q u}{(1-u^2)^2} -y_0 + \frac{u}{\gamma^2} = 0
        \end{equation*}
        where $p = w_0^\top x_0$ and $q = \|x_0\|^2 + \|w_0\|^2$.
        \vspace{1em}
        
        \textbf{Solution:} Solve for $u$ (e.g., via Newton's method), then reconstruct $x, w, y$.
    \end{frame}
